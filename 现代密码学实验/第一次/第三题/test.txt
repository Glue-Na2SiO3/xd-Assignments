import base64

def english_score(text):
    """通过英文字符频率计算文本“合理性”得分，得分越高越可能是英文明文"""
    char_frequency = {
        'a': 0.08167, 'b': 0.01492, 'c': 0.02782, 'd': 0.04253,
        'e': 0.12702, 'f': 0.02228, 'g': 0.02015, 'h': 0.06094,
        'i': 0.06094, 'j': 0.00153, 'k': 0.00772, 'l': 0.04025,
        'm': 0.02406, 'n': 0.06749, 'o': 0.07507, 'p': 0.01929,
        'q': 0.00095, 'r': 0.05987, 's': 0.06327, 't': 0.09056,
        'u': 0.02758, 'v': 0.00978, 'w': 0.02360, 'x': 0.00150,
        'y': 0.01974, 'z': 0.00074, ' ': 0.15000  # 空格频率权重较高
    }
    return sum(char_frequency.get(chr(c).lower(), 0) for c in text)

def single_char_xor(data, key_char):
    """单字节异或：字节串与单个密钥字节（整数）异或，返回结果字节串"""
    return bytes([byte ^ key_char for byte in data])

def repeating_key_xor(ciphertext, key):
    """重复密钥异或：字节串与密钥（字节串）循环异或，返回结果字节串"""
    return bytes([ciphertext[i] ^ key[i % len(key)] for i in range(len(ciphertext))])

def hamming_distance(b1, b2):
    """计算两个字节串的汉明距离（不同二进制位的数量）"""
    return sum(bin(x ^ y).count('1') for x, y in zip(b1, b2))

def break_single_byte_xor(ciphertext):
    """破解单字节异或：返回得分最高的{密钥, 解密结果, 得分}"""
    candidates = [
        {
            'key': key,
            'decrypted': single_char_xor(ciphertext, key),
            'score': english_score(single_char_xor(ciphertext, key))
        }
        for key in range(256)  # 遍历所有可能的单字节密钥（0-255）
    ]
    return max(candidates, key=lambda x: x['score'])

def find_key_size(ciphertext):
    """通过汉明距离推测最可能的密钥长度（范围：2-40）"""
    distances = []
    for key_size in range(2, 41):
        chunks = [ciphertext[i:i+key_size] for i in range(0, len(ciphertext), key_size)]
        # 计算相邻块的平均汉明距离（归一化到每字节）
        pairs = list(zip(chunks, chunks[1:]))
        if not pairs:
            continue
        avg_distance = sum(hamming_distance(p1, p2) / key_size for p1, p2 in pairs) / len(pairs)
        distances.append((key_size, avg_distance))
    # 选择平均汉明距离最小的密钥长度（更可能是正确长度）
    return min(distances, key=lambda x: x[1])[0]

def break_repeating_key_xor(ciphertext):
    """破解重复密钥异或：返回解密明文与密钥"""
    key_size = find_key_size(ciphertext)
    key = bytearray()
    # 按密钥长度分组，每组用“单字节异或破解”求对应密钥位
    for i in range(key_size):
        block = bytearray([ciphertext[j] for j in range(i, len(ciphertext), key_size)])
        key.append(break_single_byte_xor(block)['key'])
    # 用找到的密钥解密完整密文
    decrypted_message = repeating_key_xor(ciphertext, key)
    return decrypted_message, bytes(key)
